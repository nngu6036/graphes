#!/usr/bin/env bash

# ==============================================================================
# FINAL EXPLICIT GDSM EXPERIMENTS
# ==============================================================================
#
# Datasets:
#   1. Community-small
#   2. Ego-small
#   3. QM9
#   4. ZINC
#
# Seeds:
#   41, 42, 43, 44, 45
#
# Experimental protocol:
#   - Community-small / Ego-small: generate 1,024 graphs per seed.
#   - QM9 / ZINC: generate 10,000 molecules per seed.
#   - Graphlets: g3, g4, g5 (g345 experiment configuration).
#   - GDSM eigendecomposition backend: CPU.
#   - Training / generation device: cuda:0.
#   - Synthetic evaluation: GraphRNN MMD protocol + ORCA.
#   - Molecular evaluation: raw-valid metrics, NSPDK, FCD, QED.
#
# IMPORTANT:
#   - All commands are explicit; no seed loop is used.
#   - All experiment results present in the original source file are retained.
#   - Recorded outputs are stored in Bash no-op heredoc blocks:
#
#         : <<'RESULTS_...'
#         ...
#         RESULTS_...
#
#     Therefore the results remain readable in this file but are NOT executed.
#   - If no result was present for a dataset/seed in the source file, the script
#     says "No recorded result in source file"; no metric is fabricated.
#
# ==============================================================================

# ##############################################################################
# DATASET 1/4: COMMUNITY-SMALL
# ##############################################################################

# ==============================================================================
# Community-small — SEED 41
# ==============================================================================

# ------------------------------------------------------------------------------
# 1. Train
# ------------------------------------------------------------------------------
PYTHONHASHSEED=41 GDSM_EIGH_BACKEND=cpu PYTHONPATH=src \
python scripts/run_gdsm_simple_baseline.py \
  --stage train \
  --dataset community_small \
  --no-common-config \
  --wrapper-config configs/experiments/gdsm_final_explicit/community_small_seed_41.yaml \
  --dataset-root outputs/datasets \
  --serialized-dataset sbm \
  --output-root outputs/baselines \
  --seed-id 41 \
  --run-id seed_41_gdsm_final_g345 \
  --device cuda:0

# ------------------------------------------------------------------------------
# 2. Generate 1,024 graphs
# ------------------------------------------------------------------------------
PYTHONHASHSEED=41 GDSM_EIGH_BACKEND=cpu PYTHONPATH=src \
python scripts/run_gdsm_simple_baseline.py \
  --stage generate \
  --dataset community_small \
  --no-common-config \
  --wrapper-config configs/experiments/gdsm_final_explicit/community_small_seed_41.yaml \
  --dataset-root outputs/datasets \
  --serialized-dataset sbm \
  --output-root outputs/baselines \
  --seed-id 41 \
  --run-id seed_41_gdsm_final_g345 \
  --device cuda:0 \
  --generation-seed 41 \
  --generation-id seed_41_n_1024 \
  --num-samples 1024

# ------------------------------------------------------------------------------
# 3. Evaluate topology
# ------------------------------------------------------------------------------
PYTHONHASHSEED=41 ORCA_EXEC=/home/quang/orca/orca.out PYTHONPATH=src \
python scripts/evaluate_graph_generation_report.py \
  --config configs/experiments/baselines/community_small_evaluation.yaml \
  --generated-dir outputs/baselines/gdsm_simple/community_small/seed_41_gdsm_final_g345/generations/seed_41_n_1024 \
  --generated-graphs outputs/baselines/gdsm_simple/community_small/seed_41_gdsm_final_g345/generations/seed_41_n_1024/base_graphs.pkl \
  --base-graphs outputs/baselines/gdsm_simple/community_small/seed_41_gdsm_final_g345/generations/seed_41_n_1024/initial_graphs.pkl \
  --generated-stage gdsm_final \
  --reference-split test \
  --generic-mmd-protocol graphrnn \
  --num-samples 1024 \
  --output-dir outputs/baselines/gdsm_simple/community_small/seed_41_gdsm_final_g345/generations/seed_41_n_1024/evaluation_topology
