    --reference-split val \
    --output-dir "$GEN/evaluation_val"

  PYTHONPATH=src python scripts/evaluate_induced_graphlets.py \
    --config "$CFG" \
    --generated-graphs "$GEN/topology_refined_graphs.pkl" \

    --json-out "$GEN/evaluation_val/induced_graphlets345.json"
done
