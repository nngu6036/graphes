# Included experiments only:
#   - Community-small: seeds 42, 43, 44
#   - Ego-small:       seeds 42, 43, 44

# Notes:
#   - No missing scripts or seeds have been added.
#   - Each run trains, generates 1,024 graphs, and evaluates generation quality.
#   - All recorded experiment results are retained verbatim in non-executing
#     heredoc blocks so this file remains valid Bash.

export GDSM=/home/quang/GDSM
export GDSM_PYTHON=/home/quang/miniconda3/envs/defog/bin/python


# ##############################################################################
# DATASET 1/2: COMMUNITY-SMALL
# ##############################################################################


# ==============================================================================
# Community-small — Seed 42
# ==============================================================================

RUN=seed_42
N=1024

# ------------------------------------------------------------------------------
# 1. Train
# ------------------------------------------------------------------------------
PYTHONPATH=src python scripts/run_gdsm_baseline.py \
  --stage train \
  --dataset community_small \
  --common-config configs/baselines/common_community_small.yaml \
  --wrapper-config configs/baselines/gdsm_community_small.yaml \
  --seed-id 42 \
  --run-id "$RUN" \
  --device gpu

# ------------------------------------------------------------------------------
# 2. Generate
# ------------------------------------------------------------------------------
PYTHONPATH=src python scripts/run_gdsm_baseline.py \
  --stage generate \
  --dataset community_small \
  --common-config configs/baselines/common_community_small.yaml \
  --wrapper-config configs/baselines/gdsm_community_small.yaml \
  --seed-id 42 \
  --run-id "$RUN" \
  --num-samples "$N" \
  --device gpu

GEN_DIR="outputs/baselines/gdsm/community_small/$RUN/generations/seed_42_n_${N}"

# ------------------------------------------------------------------------------
# 3. Evaluate
# ------------------------------------------------------------------------------
PYTHONPATH=src python scripts/evaluate_graph_generation_report.py \
  --config configs/experiments/baselines/community_small_evaluation.yaml \
  --generated-dir "$GEN_DIR" \
  --output-dir "$GEN_DIR/evaluation_report"

# ------------------------------------------------------------------------------
# Recorded result
# ------------------------------------------------------------------------------
: <<'RESULTS_COMMUNITY_SMALL_SEED_42'
Graph-distribution MMD against held-out test graphs (lower is better)
Comparison                        Degree MMD    Clustering MMD     Orbit MMD    Spectral MMD    Graphlet MMD
train_to_test                       0.004733          0.009889      0.000399        0.020486        0.010610
gdsm_to_test                        0.018339          0.017282      0.007639        0.013240        0.018920
RESULTS_COMMUNITY_SMALL_SEED_42
