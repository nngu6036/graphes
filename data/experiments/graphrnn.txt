export GRAPHRNN=/home/quang/GraphRNN
export GRAPHRNN_PYTHON=/home/quang/miniconda3/envs/defog/bin/python

N=1024
DEVICE="gpu"

for SEED in {42..44}; do
  echo "========================================"
  echo "Running GraphRNN for SEED=$SEED"
  echo "========================================"

  RUN="seed_${SEED}"

  PYTHONHASHSEED="$SEED" PYTHONPATH=src \
  python scripts/run_graphrnn_baseline.py \
    --stage train \
    --dataset community_small \
    --common-config configs/baselines/common_community_small.yaml \
    --wrapper-config configs/baselines/graphrnn_community_small.yaml \
    --seed-id "$SEED" \
    --run-id "$RUN" \
    --device "$DEVICE" || break

  PYTHONHASHSEED="$SEED" PYTHONPATH=src \
  python scripts/run_graphrnn_baseline.py \
    --stage generate \
    --dataset community_small \
    --common-config configs/baselines/common_community_small.yaml \
    --wrapper-config configs/baselines/graphrnn_community_small.yaml \
    --seed-id "$SEED" \
    --run-id "$RUN" \
    --num-samples "$N" \
    --device "$DEVICE" || break

  GEN_DIR="outputs/baselines/graphrnn/community_small/$RUN/generations/seed_${SEED}_n_${N}"

  PYTHONHASHSEED="$SEED" PYTHONPATH=src \
  python scripts/evaluate_graph_generation_report.py \
    --config configs/experiments/baselines/community_small_evaluation.yaml \
    --generated-dir "$GEN_DIR" \
    --output-dir "$GEN_DIR/evaluation_report" || break

  echo "Completed SEED=$SEED"
done
