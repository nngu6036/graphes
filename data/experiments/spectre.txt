1. Community-small


export SPECTRE="/home/quang/SPECTRE"
export SPECTRE_PYTHON="$(command -v python)"
N=1024
DEVICE="gpu"

for SEED in {42..44}; do
  echo "========================================"
  echo "Running SPECTRE community_small for SEED=$SEED"
  echo "========================================"

  RUN="seed_${SEED}"

  # Train
  PYTHONHASHSEED="$SEED" PYTHONPATH=src \
  python scripts/run_spectre_baseline.py \
    --stage train \
    --dataset community_small \
    --wrapper-config configs/baselines/spectre_community_small.yaml \
    --seed-id "$SEED" \
    --run-id "$RUN" \
    --device "$DEVICE" \
    --overwrite || break

  # Generate
  PYTHONHASHSEED="$SEED" PYTHONPATH=src \
  python scripts/run_spectre_baseline.py \
    --stage generate \
    --dataset community_small \
    --wrapper-config configs/baselines/spectre_community_small.yaml \
    --seed-id "$SEED" \
    --run-id "$RUN" \
    --num-samples "$N" \
    --device "$DEVICE" || break

  # Evaluate
  GEN_DIR="outputs/baselines/spectre/community_small/$RUN/generations/seed_${SEED}_n_${N}"

  PYTHONHASHSEED="$SEED" PYTHONPATH=src \
  python scripts/evaluate_graph_generation_report.py \
    --config configs/experiments/baselines/community_small_evaluation.yaml \
    --generated-dir "$GEN_DIR" \
    --generated-stage spectre \
    --output-dir "$GEN_DIR/evaluation_report" || break

  echo "Completed SEED=$SEED"
done

echo "SPECTRE community_small loop finished."
